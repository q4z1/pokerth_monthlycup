User-agent: *

# Verwaltung. Deckt auch die Legacy-Weiterleitungen ab (/admin/signup,
# /admin/award/edit, /admin/upload/finaltable ...), die auf ihre jeweils
# auth-geschuetzten Gegenstuecke zeigen.
Disallow: /admin

# Anmeldung und Abmeldung tragen keinen Inhalt
Disallow: /login$
Disallow: /main/logout

# Nutzereinstellungen, keine Inhaltsseite
Disallow: /table-settings

# Health-Check von Laravel
Disallow: /up$

#   /media/* /res/* /storage/*
