# monthlycup.pokerth.net — PokerTH Monthly Cup
#
# Neu gefasst am 2026-08-18, nachdem die Seite auf Laravel umgestellt wurde.
# Die Vorgaenger-Fassung von heute Nachmittag sperrte /admin/ und /ajax/ — das
# waren die Controller der alten Eigenbau-App und existieren nicht mehr.
#
# Regeln abgeleitet aus "php artisan route:list": 30 oeffentliche gegen
# 7 login-pflichtige GET-Routen.

User-agent: *

# Verwaltung. Deckt auch die Legacy-Weiterleitungen ab (/admin/signup,
# /admin/award/edit, /admin/upload/finaltable ...), die auf ihre jeweils
# auth-geschuetzten Gegenstuecke zeigen.
Disallow: /admin

# Anmeldung und Abmeldung tragen keinen Inhalt
Disallow: /login$
Disallow: /main/logout

# Health-Check von Laravel
Disallow: /up$

# Bewusst offen und nicht gesperrt:
#
#   /results /results/cup/* /results/rankings /results/points
#   /results/halloffame /results/series /registration /signups
#   /media/* /res/* /storage/*
#
# ebenso die alten /main/...-URLs: die leiten auf ihre neuen Entsprechungen
# weiter, und ein Crawler muss ihnen folgen duerfen, damit er die neuen
# Adressen ueberhaupt lernt. Sperren wuerde die Umstellung verschleppen.
#
# Nachtrag 2026-08-18:
#
# /table-settings war gesperrt und als "Nutzereinstellungen" gefuehrt. Das
# ist eine Fehleinschaetzung: die Seite beschreibt die Spieleinstellungen der
# Cup-Tische (Blinds, Startkapital, Timeouts) und ist oeffentlicher Inhalt,
# nach dem auch gesucht wird. Die Sperre ist entfernt.
#
# Die sitemap.xml wird jetzt dynamisch aus den Routen und den vorhandenen
# Saisons erzeugt (SitemapController), enthaelt also automatisch jeden Cup
# mit Ergebnissen.

Sitemap: https://monthlycup.pokerth.net/sitemap.xml
